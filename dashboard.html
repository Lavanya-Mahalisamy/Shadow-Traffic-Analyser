// ===== 1. Read Analysis Result ===== let result = JSON.parse(localStorage.getItem("analysisResult")); const data = JSON.parse(localStorage.getItem("trafficData")); // ===== 2. If analysisResult is not available, // calculate it from Home page input ===== if (!result && data) { const networkScore = data.network === "Public Network" ? 5 : data.network === "Mobile Data" ? 3 : 2; const activityScore = data.activity === "Unknown Background Activity" ? 6 : data.activity === "Gaming" ? 4 : data.activity === "Streaming" ? 3 : 1; // Risk calculation const riskValue = networkScore + activityScore; // Traffic calculation const activeTraffic = networkScore * 20; const shadowTraffic = activityScore * 100; const totalTraffic = activeTraffic + shadowTraffic; // Risk level let riskLevel = "LOW"; if (riskValue >= 9) { riskLevel = "HIGH"; } else if (riskValue >= 6) { riskLevel = "MEDIUM"; } // Store result result = { activeTraffic: activeTraffic, shadowTraffic: shadowTraffic, totalTraffic: totalTraffic, riskLevel: riskLevel }; localStorage.setItem( "analysisResult", JSON.stringify(result) ); } // ===== 3. Display Result ===== if (result) { document.getElementById("activeTrafficValue").innerText = result.activeTraffic + " Packets"; document.getElementById("shadowTrafficValue").innerText = result.shadowTraffic + " Packets"; document.getElementById("totalTrafficValue").innerText = result.totalTraffic + " Packets"; // Device Status const deviceStatus = result.riskLevel === "HIGH" ? "RISKY" : result.riskLevel === "MEDIUM" ? "WARNING" : "Normal"; document.getElementById("deviceStatusValue").innerText = deviceStatus; // ===== 4. Color Coding ===== const activeCard = document.getElementById("activeTrafficCard"); const shadowCard = document.getElementById("shadowTrafficCard"); const totalCard = document.getElementById("totalTrafficCard"); const statusCard = document.getElementById("deviceStatusCard"); // Remove previous colors activeCard.classList.remove("green", "yellow", "red"); shadowCard.classList.remove("green", "yellow", "red"); totalCard.classList.remove("green", "yellow", "red"); statusCard.classList.remove("green", "yellow", "red"); // Active Traffic if (result.activeTraffic > 50) { activeCard.classList.add("yellow"); } else { activeCard.classList.add("green"); } // Shadow Traffic if (result.shadowTraffic > 500) { shadowCard.classList.add("red"); } else if (result.shadowTraffic > 300) { shadowCard.classList.add("yellow"); } else { shadowCard.classList.add("green"); } // Total Traffic if (result.totalTraffic > 600) { totalCard.classList.add("red"); } else if (result.totalTraffic > 400) { totalCard.classList.add("yellow"); } else { totalCard.classList.add("green"); } // Device Status if (result.riskLevel === "HIGH") { statusCard.classList.add("red"); } else if (result.riskLevel === "MEDIUM") { statusCard.classList.add("yellow"); } else { statusCard.classList.add("green"); } } else { // No data available document.getElementById("activeTrafficValue").innerText = "No Data"; document.getElementById("shadowTrafficValue").innerText = "No Data"; document.getElementById("totalTrafficValue").innerText = "No Data"; document.getElementById("deviceStatusValue").innerText = "No Analysis"; }
